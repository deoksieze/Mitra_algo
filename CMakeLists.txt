cmake_minimum_required(VERSION 3.16)
project(SymmetryDetection LANGUAGES CXX)

# Historical MATLAB/libigl sources remain excluded from these explicit targets.
add_library(symmetry_core src/core/math.cpp src/core/prepare.cpp src/core/pipeline.cpp)
target_include_directories(symmetry_core PUBLIC include)
target_compile_features(symmetry_core PUBLIC cxx_std_17)
add_library(symmetry_cpu
  src/cpu/backend.cpp src/cpu/curvature.cpp src/cpu/selection.cpp
  src/cpu/pairing.cpp src/cpu/transformations.cpp src/cpu/clustering.cpp
  src/cpu/verification.cpp)
target_link_libraries(symmetry_cpu PUBLIC symmetry_core)
add_library(symmetry_io src/io/mesh.cpp src/io/export.cpp)
target_link_libraries(symmetry_io PUBLIC symmetry_core)
foreach(target symmetry_core symmetry_cpu symmetry_io)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_library(Symmetry::core ALIAS symmetry_core)
add_library(Symmetry::cpu ALIAS symmetry_cpu)
add_library(Symmetry::io ALIAS symmetry_io)
add_executable(symmetry_cpu_cli src/app/main.cpp)
target_link_libraries(symmetry_cpu_cli PRIVATE Symmetry::cpu Symmetry::io)
include(CTest)
if(BUILD_TESTING)
  add_executable(symmetry_geometry_test tests/geometry.cpp)
  target_link_libraries(symmetry_geometry_test PRIVATE Symmetry::core)
  add_test(NAME geometry COMMAND symmetry_geometry_test)
  add_executable(symmetry_storage_test tests/storage_backend.cpp)
  target_link_libraries(symmetry_storage_test PRIVATE Symmetry::cpu)
  add_test(NAME storage_backend COMMAND symmetry_storage_test)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipeline.py
      $<TARGET_FILE:symmetry_cpu_cli> ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
endif()
